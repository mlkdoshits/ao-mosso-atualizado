const socket = io(); const btnSim = document.getElementById('btn-sim-clique'); const btnNao = document.getElementById('btn-nao'); const btnHora = document.getElementById('btn-hora-clique'); const areaBotoes = document.getElementById('area-botoes'); const horarioInput = document.getElementById('horario-input'); const statusDiv = document.getElementById('status'); // Recebe o placar sincronizado do servidor ao conectar ou atualizar socket.on('atualizar-placar', (placar) => { document.getElementById('qtd-sim').innerText = placar.sim; document.getElementById('qtd-nao').innerText = placar.nao; }); horarioInput.addEventListener('input', (e) => { let valor = e.target.value.replace(/\D/g, ""); if (valor.length > 2) { valor = valor.substring(0, 2) + ":" + valor.substring(2, 4); } e.target.value = valor; }); function fugirDoAmigo(evento) { if(evento) evento.preventDefault(); const larguraContainer = areaBotoes.offsetWidth - btnNao.offsetWidth; const alturaContainer = areaBotoes.offsetHeight - btnNao.offsetHeight; let novoX = Math.random() * larguraContainer; let novoY = Math.random() * alturaContainer; if (novoX < 160) { novoX = 160 + (Math.random() * (larguraContainer - 160)); } if (novoY > 60) { novoY = Math.random() * 60; } btnNao.style.right = 'auto'; btnNao.style.left = novoX + 'px'; btnNao.style.top = novoY + 'px'; } btnNao.addEventListener('mouseover', fugirDoAmigo); btnNao.addEventListener('pointerdown', fugirDoAmigo); btnNao.addEventListener('touchstart', fugirDoAmigo); btnSim.addEventListener('click', () => { socket.emit('resposta', 'SIM'); statusDiv.innerHTML = "🎉 🥳 SÓ VAI! AO MOSSAR LIBERADO! 🍟🍕🍔"; statusDiv.style.color = "#155724"; statusDiv.style.backgroundColor = "#d4edda"; statusDiv.style.border = "1px solid #c3e6cb"; resgatarBotaoNao(); }); btnNao.addEventListener('click', (e) => { e.preventDefault(); const chance = Math.random() * 100; if (chance <= 10) { socket.emit('resposta', 'NAO'); statusDiv.innerHTML = "😭 ❌ FOME CANCELADA! DEU RUIM... 💔🫠"; statusDiv.style.color = "#721c24"; statusDiv.style.backgroundColor = "#f8d7da"; statusDiv.style.border = "1px solid #f5c6cb"; resgatarBotaoNao(); } else { fugirDoAmigo(e); statusDiv.innerHTML = "⚠️ O botão de não falhou! Tente de novo (se conseguir) 🏃‍♂️💨"; statusDiv.style.color = "#856404"; statusDiv.style.backgroundColor = "#fff3cd"; statusDiv.style.border = "1px solid #ffeeba"; } }); btnHora.addEventListener('click', () => { const hora = horarioInput.value.trim(); if(hora.length === 5) { const partes = hora.split(':'); const h = parseInt(partes[0], 10); const m = parseInt(partes[1], 10); if (h >= 0 && h <= 23 && m >= 0 && m <= 59) { socket.emit('resposta', 'HORARIO:' + hora); statusDiv.innerHTML = "⏰ Horário marcado para às " + hora + "! ✨"; statusDiv.style.color = "#856404"; statusDiv.style.backgroundColor = "#fff3cd"; statusDiv.style.border = "1px solid #ffeeba"; horarioInput.value = ''; resgatarBotaoNao(); } else { statusDiv.innerHTML = "⚠️ Loló, insira um horário que exista de verdade!"; statusDiv.style.color = "#721c24"; statusDiv.style.backgroundColor = "#f8d7da"; statusDiv.style.border = "1px solid #f5c6cb"; } } else { statusDiv.innerHTML = "⚠️ Insira o horário completo (Ex: 12:30)!"; statusDiv.style.color = "#333"; statusDiv.style.backgroundColor = "transparent"; statusDiv.style.border = "none"; } }); function resgatarBotaoNao() { btnNao.style.left = 'auto'; btnNao.style.top = '15px'; btnNao.style.right = '40px'; }
